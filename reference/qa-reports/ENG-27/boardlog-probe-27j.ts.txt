import { FirstTimerRun, FT_MIDDLE_EASY, SKILLS } from './firsttimer.ts';
import { driftX } from '../../src/game/river.ts';
import type { Platform } from '../../src/game/types.ts';
const bank = { id: 1, x: 0, y: 600, width: 1000, height: 90, type: 'solid' } as Platform;
const far = { id: 3, x: 1600, y: 600, width: 3000, height: 90, type: 'solid' } as Platform;
const go = (on: boolean, front: number, start: number) => {
  const log = { id: 2, x: front - 240, y: 628, width: 240, height: 24, type: 'drift',
    drift: { x0: 1000, x1: 1600, dir: 1, speed: 60, width: 240, baseY: 628, offset: (front - 1000) / 60 } } as unknown as Platform;
  const line = Array.from({ length: 200 }, (_, i) => ({ x: 100 + i * 24, y: 600 }));
  const rows: string[] = [];
  let n = 0;
  const run = new FirstTimerRun('1-1', 1, FT_MIDDLE_EASY, { route: line, cap: 4, rescueAfter: 1000, onFrame: (sim: any) => {
    if (n++ % 3) return; const p = sim.p; const lg = sim.platforms.find((q: any) => q.id === 2);
    rows.push(`${(n/30).toFixed(1)} p${Math.round(p.x)},${Math.round(p.y)}${p.grounded ? 'g' : ''} on${sim.standingOn} log${Math.round(lg.x)} dx${Math.round(driftX(lg.drift, n/30))}`);
  } } as any);
  run.sim.bare([bank, log, far]);
  const st = (run.sim as any).e.currentStage;
  st.config.levelHeight = 700; st.rivers = [{ x0: 1000, x1: 1600, surfaceY: 640, dir: 1, speed: 60 }];
  run.sim.stand(start, 600); run.prog.relocate(run.sim.p);
  const was = SKILLS.boardLog; SKILLS.boardLog = on;
  try { const r = run.play(); return JSON.stringify(r.softFalls.map((f: any) => [Math.round(f.t*10)/10, Math.round(f.x)])) ; } finally { SKILLS.boardLog = was; }
};
for (const [front, start] of [[960,700],[975,700],[986,700],[1000,700],[1015,700],[930,500],[950,500]])
console.log(front, start, "ON", go(true, front, start), "OFF", go(false, front, start));
