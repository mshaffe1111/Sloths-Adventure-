import json, sys
n, S = sys.argv[1], sys.argv[2]
def load(p):
    L = [json.loads(l) for l in open(p)]; return L[:-1], L[-1]
B, bs = load(f'{S}/tr/{n}-base.jsonl'); H, hs = load(f'{S}/tr/{n}-head.jsonl')
def lives(T):
    out = []; lastRead = None
    for i in range(1, len(T)):
        if T[i].get('bl'): lastRead = T[i]['t']
        if T[i]['l'] < T[i-1]['l']:
            j = i - 1
            while j > 0 and T[j]['on'] == '-': j -= 1
            # how he left it: a jump started within 3 frames of leaving (or in the air before the fall) or a walk-off
            jumped = any(T[k].get('jmp') for k in range(j - 2, i))
            out.append(dict(t=T[i]['t'], x=round(T[i-1]['x']), off=T[j]['on'].split(',')[0], vx=T[j]['vx'], how='jump' if jumped else 'walk-off', rd=round(T[i]['t'] - lastRead, 1) if lastRead else None))
    return out
rx = hs['rescueXs'][0]
bl, hl = lives(B), lives(H)
near = lambda L: [l for l in L if abs(l['x'] - rx) < 700]
fmt = lambda L: '; '.join(f"x {l['x']} {l['how']} off {l['off']} (read {l['rd']}s before)" for l in L)
print(f"{n} rescue x {rx}: head {len(near(hl))} lives there: {fmt(near(hl))}")
print(f"   base {len(near(bl))} lives there: {fmt(near(bl))}")
print(f"   head all lives by how: walk-off {sum(l['how']=='walk-off' for l in hl)}, jump {sum(l['how']=='jump' for l in hl)} | base walk-off {sum(l['how']=='walk-off' for l in bl)}, jump {sum(l['how']=='jump' for l in bl)}")
