# usage: r28diff-27h.py stage base.jsonl oldhead.jsonl newhead.jsonl: rule 28 for SKILLS.boardHop (ENG-27h). base = ENG-27g's rows with both its
# skills off; oldhead = ENG-27g's with both on; newhead = both on and boardHop. Flips and tier counts are read base -> newhead (the lead's bar);
# what boardHop itself moved is read oldhead -> newhead, and every such row must have met a log across a lip in jumpFor (bhReads)
import json, sys
def load(p):
    d = {}
    for l in open(p):
        if l.strip(): r = json.loads(l); d[(r['bot'], r['setting'], r['seed'])] = r
    return d
st, base, old, new = sys.argv[1], load(sys.argv[2]), load(sys.argv[3]), load(sys.argv[4])
K = ['end', 'pass', 'time', 'goal', 'rescues', 'turnBacks', 'hits', 'lives', 'hitBy', 'cast', 'stuck']
same = lambda a, b: all(a.get(x) == b.get(x) for x in K)
miss = [k for k in base if k not in new]; errs = [k for k in new if 'error' in new[k]]
if miss or errs: print('MISSING', miss[:5], 'ERRORS', errs[:5])
groups = []
for k in base:
    if (k[0], k[1]) not in groups: groups.append((k[0], k[1]))
flips = []; mvOld = []; drops = []
for g in groups:
    ks = sorted(k for k in base if (k[0], k[1]) == g)
    pb, po, pn = (sum(d[k]['pass'] for k in ks) for d in (base, old, new))
    mo = [k for k in ks if not same(old[k], new[k])]; mvOld += mo
    fl = [f"{k[2]}:{'P' if base[k]['pass'] else 'F'}->{'P' if new[k]['pass'] else 'F'}" for k in ks if base[k]['pass'] != new[k]['pass']]
    flips += [(k, base[k]['pass'], new[k]['pass']) for k in ks if base[k]['pass'] != new[k]['pass']]
    if pn < pb: drops.append(g)
    lb, ln = sum(base[k]['lives'] for k in ks), sum(new[k]['lives'] for k in ks)
    print(f"{g[0]:6} {g[1]:6} pass base {pb}/40, 27g head {po}/40, new head {pn}/40; lives {lb} -> {ln}; boardHop moved {len(mo):3}; flips vs base {','.join(fl) or '-'}")
pf = [f for f in flips if f[1] and not f[2]]
print(f"{st}: vs base P->F {len(pf)} {[f'{k[0]} {k[1]} {k[2]}' for k, _, _ in pf]}, F->P {sum(1 for f in flips if not f[1] and f[2])}; tier counts dropped: {drops or 'none'}")
un = [k for k in mvOld if not new[k].get('bhReads')]
print(f"{st}: boardHop moved {len(mvOld)} of {len(base)} rows (27g head -> new); with no log across a lip in jumpFor: {un or 'none'}")
pf27 = [k for k in base if base[k]['pass'] and not old[k]['pass']]
print(f"{st}: the 8 traced flips (27g head) now: {[(f'{k[0]} {k[1]} {k[2]}', 'P' if new[k]['pass'] else 'F') for k in pf27]}")
