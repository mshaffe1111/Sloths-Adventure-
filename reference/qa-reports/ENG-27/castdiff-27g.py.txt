# usage: python3 castdiff.py base.jsonl new.jsonl [title]  -> per bot/setting: rows moved, pass counts, flips; then the moved rows
import json, sys
def load(p):
    d = {}
    for l in open(p):
        l = l.strip()
        if l: r = json.loads(l); d[(r['bot'], r['setting'], r['seed'])] = r
    return d
a, b = load(sys.argv[1]), load(sys.argv[2])
if len(sys.argv) > 3: print(sys.argv[3])
keys = ['end', 'pass', 'time', 'goal', 'rescues', 'turnBacks', 'hits', 'lives', 'hitBy', 'cast', 'stuck']
order = []
for k in list(a) + list(b):
    g = (k[0], k[1])
    if g not in order: order.append(g)
tot = moved_all = 0; lines = []
for g in order:
    ks = sorted({k for k in list(a) + list(b) if (k[0], k[1]) == g}, key=lambda k: k[2])
    mv = 0; flips = []; pa = pb = 0; miss = 0
    for k in ks:
        ra, rb = a.get(k), b.get(k)
        if ra is None or rb is None: miss += 1; continue
        pa += ra['pass']; pb += rb['pass']
        if any(ra.get(x) != rb.get(x) for x in keys):
            mv += 1
            lines.append(f"  {k[0]:6} {k[1]:6} {k[2]:2}  {'P' if ra['pass'] else 'F'}->{'P' if rb['pass'] else 'F'}  time {ra['time']} -> {rb['time']}  rescues {ra['rescues']}->{rb['rescues']}" + (f"  stuck {rb['stuck']}" if ra['stuck'] != rb['stuck'] else ''))
        if ra['pass'] != rb['pass']: flips.append(f"{k[2]}:{'P' if ra['pass'] else 'F'}->{'P' if rb['pass'] else 'F'}")
    tot += len(ks); moved_all += mv
    print(f"{g[0]:6} {g[1]:6}  rows {len(ks)} moved {mv:3}  pass {pa}/{len(ks)} -> {pb}/{len(ks)}  flips {','.join(flips) or '-'}" + (f"  MISSING {miss}" if miss else ''))
print(f"total moved {moved_all} of {tot}")
print('\n'.join(lines))
