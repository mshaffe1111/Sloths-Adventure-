#!/bin/bash
# easy.sh <tree> <label> <pace param %>: middle + ride-only on every Easy race (home 1-3, post 4-5)
T=$1; L=$2; P=$3; S=$(dirname "$0")
cd "$T"
for tier in middle ride; do for r in "home easy 1" "home easy 2" "home easy 3" "post easy 4" "post easy 5"; do
  npx tsx $S/table-27w.mts "$T" "$L" $tier $r 72 $P 2>&1 | tail -1
done; done
