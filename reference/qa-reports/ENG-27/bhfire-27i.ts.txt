// and the same row with NO_HOP=1 run by the caller; a life is boardHop's only if a fire came within 3 s before it
import { CAST_BOTS } from './castBots.ts';
import { firstTimer } from './firsttimer.ts';
const stage = process.argv[2];
for (const spec of process.argv.slice(3)) {
  const [bot, setting, seedS] = spec.split(':');
  (globalThis as any).__ft = []; (globalThis as any).__bl = []; (globalThis as any).__bh = []; (globalThis as any).__bhf = []; (globalThis as any).__up = [];
  let lives0: number | undefined; const lost: any[] = [];
  const r = firstTimer(stage, Number(seedS), (CAST_BOTS as any)[bot][setting], { cap: 1200, dotSetting: true, onFrame: (sim: any) => {
    const e = sim.e; if (lives0 !== undefined && e.lives < lives0) lost.push({ t: Math.round(sim.t * 10) / 10, x: Math.round(e.player.x) }); lives0 = e.lives;
  } } as never);
  const fires = (globalThis as any).__bhf as any[];
  console.log(JSON.stringify({ stage, bot, setting, seed: Number(seedS), pass: r.end === 'shrine' && r.rescues === 0, end: r.end, lost, fires, ups: (globalThis as any).__up,
    after: lost.filter((l) => fires.some((f) => l.t - f.t >= 0 && l.t - f.t < 3)) }));
}
