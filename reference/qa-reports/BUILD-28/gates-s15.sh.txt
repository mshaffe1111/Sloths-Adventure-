#!/bin/bash
# BUILD-28 box 12: the ONE gate run on frozen cb7b9af11 ($1 = worktree), one gate at a time; PASS/FAIL lines to $S/gates.txt
W=$1; S=$2; cd "$W" || exit 9
O=$S/gates.txt; : > "$O"
g() { local name=$1; shift; local t0=$(date +%s); bash -c "$*" > "$S/g-$name.log" 2>&1; local rc=$?; local t=$(( $(date +%s) - t0 ));
  if [ $rc -eq 0 ]; then echo "PASS $name (${t}s)" >> "$O"; else echo "FAIL $name rc=$rc (${t}s)" >> "$O"; fi; }
g tsc "npx tsc --noEmit -p ."
g generality "npx tsx tools/engine/generalitycheck.ts"
CI=$(sed -n 's/^ *- run: \(npx tsx --test tools\/engine\/library.test.ts.*\)$/\1/p' .github/workflows/gates.yml)
g ciline "$CI"
for i in 1 2 3 4 5 6; do
  g suite$i "npx tsx --test --test-shard=$i/6 src/audio/*.test.ts src/game/*.test.ts src/components/*.test.ts src/components/**/*.test.ts src/game/gen2/*.test.ts src/game/gen2/worlds/*.test.ts src/game/art/*.test.ts"
done
g seedlists 'L=$(npx tsx tools/engine/registrycheck.ts --lists) || exit 1; f=0; for s in $L; do echo "== $s"; npx tsx tools/engine/seedlist.ts "$s" --check || { echo "FAILED $s"; f=1; }; done; exit $f'
g pingrep 'npx tsx --test $(grep -rlE "setPick11|roll11Seed|provenSeedFor|firstTimer\(|castRun\(|RealPlay" src tools --include=*.test.ts)'
g tierbots "npx tsx --test tools/inspect/tierBots.test.ts"
g drawcheck "node tools/perf/drawcheck.mjs"
g clipcheck "node tools/perf/clipcheck.mjs"
echo "ALL DONE" >> "$O"
