const T = process.env.TREE!;
await import(T + '/src/game/headlessDom.ts');
const { generateAllLevels } = await import(T + '/src/game/worldsData.ts');
const { listedStage, playFinish } = await import(T + '/src/game/gen2/listed.ts');
const { buildGen2Stage } = await import(T + '/src/game/gen2/pipeline.ts');
const seed = +(process.env.SEED ?? 1), d = (process.env.D ?? 'normal') as any;
const cfg = generateAllLevels().find((c: any) => c.id === '2-6' && !c.secret)!;
const row = listedStage('2-6')!;
const raw: any = buildGen2Stage({ config: cfg, kit: row.kit(), seed, difficulty: d });
const before = new Map(raw.platforms.map((p: any) => [p.id, { ...p }]));
const st: any = playFinish(row, raw as never, d);
const after = new Set(st.platforms.map((p: any) => p.id));
const R = (v: number) => Math.round(v);
for (const r of st.rivers) console.log('RIV', R(r.x0), R(r.x1), R(r.surfaceY), 'sp', r.speed, r.wheelPond ? 'pond' : '', r.safe ? 'safe' : '');
for (const f of st.gen2.falls) console.log('FALL', f.kind, R(f.x0), R(f.x1), R(f.y), f.site ?? '');
for (const [id, p] of before as any) if (!after.has(id)) console.log('DROPPED', id, R(p.x), R(p.y), R(p.width), p.height);
const legs = st.gen2.tower?.legs ?? [];
for (const g of legs) console.log('LEG', g.travel, R(g.x0), R(g.x1), R(g.y0), R(g.y1), g.node ?? g.token ?? '');
const W = (process.env.WIN ?? '').split(';').filter(Boolean).map((s) => s.split(',').map(Number));
for (const [y0, y1] of W) for (const p of raw.platforms) if (p.y >= y0 && p.y <= y1) console.log('P', p.id, R(p.x), R(p.y), R(p.width), R(p.height), p.kind ?? '', p.art ?? '', p.role ?? '', p.wheel ? 'wheel' : '', after.has(p.id) ? '' : 'GONE');
for (const n of (st.gen2.nodes ?? [])) console.log("N", JSON.stringify(n).slice(0, 220)); console.log("TOWER", JSON.stringify(Object.keys(st.gen2.tower ?? {})), JSON.stringify(Object.keys(st.gen2)));
