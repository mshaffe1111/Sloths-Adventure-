// 2-6 stills: Pokey on porch #5 (beat A), on the start floor under it, and on the climb's top bank
import fs from 'node:fs';
const PWm = await import('/opt/node-tools/node_modules/playwright/index.js'); const chromium = PWm.chromium ?? PWm.default.chromium;
const [port, out, diff = 'normal', orient = 'sideways', tag = 'x'] = process.argv.slice(2);
fs.mkdirSync(out, { recursive: true });
const b = await chromium.launch();
const E = "(typeof window.__engine === 'function' ? window.__engine() : window.__engine)";
const vp = orient === 'upright' ? { width: 412, height: 915 } : { width: 915, height: 412 };
const ctx = await b.newContext({ viewport: vp, deviceScaleFactor: 1 });
await ctx.addInitScript((d) => { try { localStorage.setItem('sloth_difficulty', d); } catch {} }, diff);
const p = await ctx.newPage();
p.on('pageerror', (e) => console.log('ERR', e.message));
await p.goto(`http://127.0.0.1:${port}/?lessons=off`);
await p.waitForTimeout(2500);
await p.evaluate(() => window.__play('2-6', 0));
await p.waitForTimeout(4000);
await p.waitForFunction((E) => { const e = eval(E); return !!(e && e.currentStage && e.player && e.platforms); }, E, { timeout: 90000 });
await p.waitForFunction(() => document.documentElement.dataset.hudCard !== '1', null, { timeout: 30000 }).catch(() => {});
await p.waitForTimeout(2000);
// LAND-26: frames on a grid (Pokey parked at each point, the camera follows), for finding a look by picture
const pts = (process.env.PTS ?? '').split(';').filter(Boolean).map((s) => s.split(',').map(Number));
const info = await p.evaluate(([E]) => { const e = eval(E); return { seed: e.currentStage.gen2?.seed, w: e.currentStage.width ?? e.levelWidth, h: e.currentStage.height ?? e.levelHeight }; }, [E]);
console.log(JSON.stringify(info));
for (const [x, y] of pts) {
  await p.evaluate(([E, x, y]) => { const e = eval(E); e.isPaused = true; for (const hh of e.currentStage?.proto?.hints ?? []) hh.shown = true;
    Object.assign(e.player, { x, y, vx: 0, vy: 0, grounded: false, invincibleTimer: 9 }); for (let i = 0; i < 30; i++) { e.player.x = x; e.player.y = y; e.player.vy = 0; e.update(1 / 30); } }, [E, x, y]);
  for (let i = 0; i < 6; i++) { await p.evaluate(([E, x, y]) => { const e = eval(E); e.player.x = x; e.player.y = y; e.player.vy = 0; e.update(1 / 30); e.render(); }, [E, x, y]); await p.evaluate(() => new Promise((r) => requestAnimationFrame(() => r(0)))); }
  await p.screenshot({ path: `${out}/${tag}-g-${x}-${y}.jpg`, type: 'jpeg', quality: 70 });
}
await b.close();
