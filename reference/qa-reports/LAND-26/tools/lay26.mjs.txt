// 2-6 stills: Pokey on porch #5 (beat A), on the start floor under it, and on the climb's top bank
import fs from 'node:fs';
const PWm = await import('/opt/node-tools/node_modules/playwright/index.js'); const chromium = PWm.chromium ?? PWm.default.chromium;
const [port, out, diff = 'normal', orient = 'sideways', tag = 'x'] = process.argv.slice(2);
fs.mkdirSync(out, { recursive: true });
const b = await chromium.launch();
const E = "(typeof window.__engine === 'function' ? window.__engine() : window.__engine)";
const vp = orient === 'upright' ? { width: 412, height: 915 } : { width: 915, height: 412 };
const ctx = await b.newContext({ viewport: vp, deviceScaleFactor: 1 });
await ctx.addInitScript((d) => { try { localStorage.setItem('sloth_difficulty', d); } catch {} }, diff);
const p = await ctx.newPage();
p.on('pageerror', (e) => console.log('ERR', e.message));
// LAND-26: the start area of 2-6's layouts 0..N-1 (spawn, and 450 px up from it), for finding a look by picture
const N = Number(process.env.N ?? 12);
for (let k = 0; k < N; k++) {
  await p.goto(`http://127.0.0.1:${port}/?lessons=off`); await p.waitForTimeout(1500);
  await p.evaluate((k) => window.__play('2-6', k), k); await p.waitForTimeout(2500);
  await p.waitForFunction((E) => { const e = eval(E); return !!(e && e.currentStage && e.player && e.platforms); }, E, { timeout: 90000 });
  await p.waitForFunction(() => document.documentElement.dataset.hudCard !== '1', null, { timeout: 30000 }).catch(() => {});
  const sp = await p.evaluate((E) => { const e = eval(E); return { x: Math.round(e.player.x), y: Math.round(e.player.y), seed: e.currentStage.gen2?.seed }; }, E);
  for (const [dx, dy] of [[0, 0], [450, -450]]) {
    const x = sp.x + dx, y = sp.y + dy;
    await p.evaluate(([E, x, y]) => { const e = eval(E); e.isPaused = true; for (const hh of e.currentStage?.proto?.hints ?? []) hh.shown = true; for (let i = 0; i < 30; i++) { Object.assign(e.player, { x, y, vx: 0, vy: 0, invincibleTimer: 9 }); e.update(1 / 30); } e.render(); }, [E, x, y]);
    await p.evaluate(() => new Promise((r) => requestAnimationFrame(() => r(0))));
    await p.screenshot({ path: `${out}/${tag}-L${String(k).padStart(2, '0')}-s${sp.seed}-${dx}.jpg`, type: 'jpeg', quality: 70 });
  }
  console.log(k, JSON.stringify(sp));
}
await b.close();
