import json, sys
# widecmp.py <maindir> <findir>: seeds 1-10, every bot, both settings: BAR PASS -> FAIL (main pass, final fail), FAIL -> PASS, moved
md, fd = sys.argv[1:3]
tot = {'pf': [], 'fp': [], 'moved': 0, 'n': 0}
for st in ['2-1', '2-2', '2-4']:
    M = {(r['bot'], r['setting'], r['seed']): r for r in map(json.loads, filter(str.strip, open(f'{md}/main-{st}.jsonl'))) if 1 <= r['seed'] <= 10}
    F = {(r['bot'], r['setting'], r['seed']): r for r in map(json.loads, filter(str.strip, open(f'{fd}/wide-{st}.jsonl')))}
    for k, f in sorted(F.items()):
        m = M.get(k); tot['n'] += 1
        if not m: print('NO MAIN ROW', st, k); continue
        if m['pass'] and not f['pass']: tot['pf'].append(f"{st} {k[0]} {k[1]} {k[2]}: main PASS {m['time']} r{m['rescues']} -> FAIL {f['time']} r{f['rescues']}")
        if not m['pass'] and f['pass']: tot['fp'].append(f"{st} {k[0]} {k[1]} {k[2]}")
        if (m['pass'], m['time'], m['rescues'], m.get('lives')) != (f['pass'], f['time'], f['rescues'], f.get('lives')): tot['moved'] += 1
    missing = [k for k in M if k not in F]
    if missing: print(st, 'missing final rows', len(missing))
print(f"rows {tot['n']}; BAR PASS -> FAIL vs main: {len(tot['pf'])}; FAIL -> PASS: {len(tot['fp'])}; moved {tot['moved']}")
for l in tot['pf']: print('  PF', l)
print('  FP', ', '.join(tot['fp']))
