// (ENG-PLANT-CI red 4, CI-9: the judged flight is caught by a block's lip hoist as the engine's walls catch him (engine.ts, the walls
//  before the floors: a solid WALL_H or taller beside him, his feet at most forgive.mantle under its top, not rising faster than 120 px/s,
//  met from outside or, with the step help, his body at most LIP_IN into its face on its near half, room on its top). 2-7 home middle Easy
//  2 and Normal 2 each came down hoisted onto the bank he hopped from. The toys: a jump straight up beside a tall block, his body 4 px into
//  its left face, flown by hopFlight and pressed on the REAL engine (a bare world, a wide floor under it): his top of the jump 20 px under
//  the block's top, the engine hoists him onto it (the judge must catch him there); 30 px under, past the mantle, neither does (the guard))
async function hoistToy(apexUnder: number) {
  const { RealPlay } = await import('../../src/game/realPlay.ts');
  const { airStep, takeOffStep, PHYS } = await import('../../src/game/physics.ts');
  const { hopFlight } = await import('./firsttimer.ts');
  const rp = new RealPlay('1-1', 1);
  const e = rp.e as unknown as { forgive: { landPx: number; mantle: number }; assistNow(): number; easy: boolean; stepHelp: boolean; standingOnId: number };
  const w = rp.p.width, assist = e.assistNow();
  // (the jump's rise, held all the way, no steer: the take-off frame and the air ticks as the judge flies them)
  const b = { x: 0, y: 0, vx: 0, vy: -PHYS.jumpV };
  let rise = 0;
  for (let i = 1; i < 75 && b.vy < 0; i++) { (i === 1 ? takeOffStep : airStep)(b, DT, 0, true); rise = Math.max(rise, -b.y); }
  const top = 3000, x0 = 5000 - 19 + 4, y0 = top + apexUnder + rise;
  // (the floor 100 px under the take-off: the judged flight stops 200 px under it)
  const block = { id: 14, x: 5000, y: top, width: 300, height: y0 + 100 - top, type: 'solid' as const };
  const floor = { id: 18, x: x0 - 1500, y: y0 + 100, width: 3000, height: 40, type: 'solid' as const };
  const h: DrawnHop = { kind: 'single', hold: 5, d2: -1, dblAt: 0, hold2: 0, upOnly: 5 };
  const others = { list: [block, floor], at: (o: { x: number }) => () => o.x, w, easy: e.easy, assist, land: e.forgive.landPx, glide: false, h: rp.p.height, mantle: e.forgive.mantle, stepHelp: e.stepHelp };
  const judged = hopFlight(floor, () => floor.x, x0, y0, 1, h, 0, 0, 24, PHYS.doubleV, undefined, others) !== null;
  rp.bare([block, floor]);
  rp.fly(x0, y0, 0, -PHYS.jumpV);
  let on = -1;
  for (let i = 0; i < 120 && on < 0; i++) {
    rp.frame({ jump: true });
    if (rp.p.grounded) on = e.standingOnId;
  }
  return { judged, on, stepHelp: e.stepHelp, mantle: e.forgive.mantle };
}
for (const [name, apexUnder, lands] of [
  ['(a) the top of his jump 20 px under the block\'s top (past the floor\'s 14 px drop catch, inside the 26 px mantle): the engine hoists him onto it, the judge must catch him there', 20, 14],
  ['(b) 30 px under, past the mantle: neither hoists him, he comes down on the floor (the guard: no hoist the engine does not give)', 30, 18],
] as const) {
  test(`the judged flight's lip hoist is the engine's (ENG-PLANT-CI red 4, CI-9): ${name}`, async () => {
    const r = await hoistToy(apexUnder);
    assert.ok(r.stepHelp && r.mantle === 26, `the built world: step help ${r.stepHelp}, mantle ${r.mantle}`);
    assert.equal(r.on, lands, `the built block: the engine set him down on #${r.on}, the case wants #${lands}`);
    assert.equal(r.judged, r.on === 18, `the judge says ${r.judged ? 'he lands on the floor #18' : 'something else catches him first'}; the engine set him down on #${r.on}`);
  });
}
