// usage: tsx toy27.mts <tree> : c27's toy, tree-agnostic. W1's signature, read by what the engine does: a stand of 6 s or more on a bank
// (not a log), then a hop off that bank within 2 s; such a hop set back by the water (one frame moving him > 60 px, no life lost) counts.
// Plus the race verdict and W1_CAP's stands past the cap (if the tree has it).
const [T] = process.argv.slice(2);
await import(`${T}/src/game/headlessDom.ts`);
const { homeRun } = await import(`${T}/tools/inspect/homeToys.ts`);
const ft = await import(`${T}/tools/inspect/firsttimer.ts`);
const { HOME_CODE, POST_CODE } = await import(`${T}/src/game/gen2/toyHome.ts`);
for (const [race, code, seed] of [['home', HOME_CODE, 1], ['post', POST_CODE, 5]] as const) {
  let piece: any, lives: any, x: any, f = 0, still = -1, stood = -1, stoodOn: number | null = null, flight: string | null = null;
  const bad: string[] = [], hops: string[] = [];
  if (ft.W1_CAP) { ft.W1_CAP.on = true; ft.W1_CAP.rows = []; }
  homeRun('middle', 'normal', seed, 1, { onFrame: (sim: any) => { const e = sim.e; const t = f++ / 30; piece ??= e.pieces.find((q: any) => q.kind === 'pacer');
    const here = e.player.grounded ? e.platforms.find((q: any) => q.id === e.standingOnId) : null;
    if (here && here.type !== 'drift' && Math.abs(e.player.vx) < 1) { if (still < 0) still = t; if (t - still >= 6) { stood = t; stoodOn = here.id; } } else still = -1;
    if (!e.player.grounded && flight === null && x !== undefined && stoodOn !== null && t - stood <= 2) { flight = `t${t.toFixed(1)} off f${stoodOn} x ${Math.round(x)}`; hops.push(flight); }
    if (x !== undefined && lives === e.lives && Math.abs(e.player.x - x) > 60 && flight) bad.push(`${flight} set back ${Math.round(x)}->${Math.round(e.player.x)}`);
    if (e.player.grounded) flight = null;
    x = e.player.x; lives = e.lives; } } as any, code);
  const v = piece?.verdict;
  console.log(`${race} N ${seed} | hops after a 6 s stand ${hops.length} | set back ${bad.length} ${bad.join(' ; ')} | ${v ? (v.won ? 'WON' : 'LOST') + ' margin ' + (v.him - v.boat).toFixed(2) : 'none'} | cap ${ft.W1_CAP ? JSON.stringify(ft.W1_CAP.rows.map((r: any) => [Math.round(r.x), +r.t0.toFixed(2), +r.dur.toFixed(2), +r.cycle.toFixed(2)])) : 'n/a'}`);
}
