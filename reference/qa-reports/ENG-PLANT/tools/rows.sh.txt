#!/bin/bash
# usage: rows.sh <frozen tree> <outdir>   castBots rows on the river stages, every bot, both settings, seeds 1-40
T=$1; O=$2; mkdir -p $O; cd $T
for st in 2-1 2-2 2-4 2-5 2-7 1-4; do
  npx tsx tools/inspect/castBots.ts --stage $st --seeds 1-40 --bots kid,adult,child,middle,expert --setting both --par 2 --rows $O/$st.jsonl > $O/$st.log 2>&1
  echo "$st exit $? $(wc -l < $O/$st.jsonl) rows $(TZ=America/New_York date +%l:%M)" >> $O/progress.txt
done
