import sys, re
# part.py <a.err> <b.err> [ctx]: the first frame where the two runs part (position or pad), with context before it
a, b = sys.argv[1:3]; ctx = int(sys.argv[3]) if len(sys.argv) > 3 else 4
fa = [l.rstrip() for l in open(a) if l.startswith('    f ')]
fb = [l.rstrip() for l in open(b) if l.startswith('    f ')]
key = lambda l: re.sub(r' meant .*? on ', ' on ', re.sub(r' stop', '', l))
for i, (x, y) in enumerate(zip(fa, fb)):
    if key(x) != key(y):
        print(f'part at frame line {i}')
        for j in range(max(0, i - ctx), min(len(fa), i + ctx)):
            print('A', fa[j][4:170]); print('B', fb[j][4:170])
        break
else: print('no parting in', min(len(fa), len(fb)), 'lines')
