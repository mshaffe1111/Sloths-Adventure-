import json,sys,collections
rows=[json.loads(l) for f in sys.argv[1:] for l in open(f) if l.startswith('{')]
by=collections.defaultdict(dict)
for r in rows: by[(r['seed'],r['tier'],r['setting'],r['v'])][r['on']]=r
seeds=sorted({k[0] for k in by})
print(f"rows {len(rows)}, seeds {seeds[0] if seeds else '-'}..{seeds[-1] if seeds else '-'}")
bite=[]
for k,d in sorted(by.items()):
  if True in d and False in d:
    a,b=d[True],d[False]
    if b['rescues']>a['rescues'] or b['lives']>a['lives'] or b['end']!=a['end']:
      bite.append((k,a,b))
for k,a,b in bite: print(k, f"ON {a['end']} {a['s']}s L{a['lives']} R{a['rescues']} | OFF {b['end']} {b['s']}s L{b['lives']} R{b['rescues']}")
ps=collections.defaultdict(lambda:[0,0,0])
for (s,t,st,v),d in by.items():
  if t!='child' and True in d and False in d:
    ps[s][0]+=d[False]['rescues']; ps[s][1]+=d[True]['rescues']; ps[s][2]+=1
print("per seed middle+expert rescues OFF/ON (runs):", {s:tuple(x) for s,x in sorted(ps.items()) if x[0] or x[1]})
