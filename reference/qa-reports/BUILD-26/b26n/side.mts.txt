const out: string[] = []; let run: any;
const P = FP.FigPush.prototype; const orig = P.freeSide;
P.freeSide = function (v: any, p: any, t: number, h: any) {
  const nw = orig.call(this, v, p, t, h);
  const tg = this.target; this.target = null; const old = orig.call(this, v, p, t, h); this.target = tg;
  if (old !== nw) out.push(`t ${t.toFixed(2)} at ${Math.round(p.x)},${Math.round(p.y)} old ${old} new ${nw} target ${tg ? tg.id : '-'}`);
  return nw;
};
