import { FirstTimerRun, FT_CHILD_EASY, FT_MIDDLE_EASY } from '/home/user/wt-toy/tools/inspect/firsttimer.ts';
// 1-4 Easy 37's lip, y less 1684: #44 360-760 @1000, #45 72-360 @960 (a thin bough 40 px up that starts AT the lip), the floor under
const plats = [
  { id: 44, x: 360, y: 1000, width: 400, height: 16, type: 'solid' },
  { id: 45, x: -428, y: 960, width: 788, height: 16, type: 'solid' },
  { id: 9, x: -1000, y: 1348, width: 3000, height: 100, type: 'solid' },
];
// the way as C reads it: one slant from the vine top (786,1000) to the far vine (90,960)
const route = Array.from({ length: 30 }, (_, i) => ({ x: 786 - i * 24, y: 1000 - (40 * i * 24) / 696 })).filter((q) => q.x >= 90);
const low = (who: any, seed: number, plant: boolean, late: boolean) => {
  let lo = 0; let g = true; let off = "";
  const run: any = new FirstTimerRun('1-1', seed, { ...who, look: 0 }, { route, cap: 6, rescueAfter: 100, onFrame: (s: any) => { lo = Math.max(lo, s.p.y); if (g && !s.p.grounded && !off) off = `off x ${Math.round(s.p.x)} vy ${Math.round(s.p.vy)} t ${(s.t ?? 0).toFixed?.(2)}`; g = s.p.grounded; } });
  run.sim.bare(plats as never, []);
  run.sim.stand(700, 1000); run.sim.p.facing = -1; run.lastSteer = -1;
  if (late) { const ep = run.edgePlan.bind(run); run.edgePlan = () => { const e = ep(); if (e.how !== 'run') e.how = 'late'; return e; }; }
  if (plant) run.stepUpAtLip = () => null;
  run.prog.relocate(run.sim.p); run.play();
  return `${Math.round(lo)} ${off}`;
};
for (const [n, who] of [['child', FT_CHILD_EASY], ['middle', FT_MIDDLE_EASY]] as const)
  for (const seed of [1, 2, 3, 4, 5, 6]) console.log(n, seed, 'rule', low(who, seed, false, true), 'plant', low(who, seed, true, true));
