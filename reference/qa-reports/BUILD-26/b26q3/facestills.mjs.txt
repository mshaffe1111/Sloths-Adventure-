// BUILD-26s (session 3 of BUILD-26q): stills of THE FACE (bde501f86) on the real page, 2-6, Easy and Normal, upright 412x915 and
// sideways 915x412, 2x. stills-at-setting.mjs's way (engine paused, Pokey placed, 60 frames stepped, then 30 stepped AND rendered),
// placed by the BUILT stage's own soft pits: gen2.drops whose ends.kind is 'floor' with a catch bough over the span (towerSoftPit.test's
// read). Pokey stands on the street before the pit's near lip, facing it. Per setting and orientation: a WIDE still (the whole phone
// screen) and a CLOSE still (the same frame cut to the pit: both lips, both faces, the bough, Pokey).
//   node facestills.mjs <port> <out dir> <layout> <pit k: 0 = the lowest soft pit (the first met climbing)> [side: near|far]
import fs from 'node:fs';
import path from 'node:path';
const { chromium } = await import(process.env.PW ?? 'playwright');
const [port, out, layoutS = '0', kS = '0', side = 'near'] = process.argv.slice(2);
const layout = +layoutS, K = +kS;
fs.mkdirSync(out, { recursive: true });
const b = await chromium.launch();
const E = "(typeof window.__engine === 'function' ? window.__engine() : window.__engine)";
for (const diff of ['easy', 'normal']) for (const [tag, w, h] of [['upright', 412, 915], ['sideways', 915, 412]]) {
  const ctx = await b.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: 2 });
  await ctx.addInitScript((d) => { try { localStorage.setItem('sloth_difficulty', d); } catch {} }, diff);
  const p = await ctx.newPage();
  p.on('pageerror', (e) => console.log('ERR', e.message));
  await p.goto(`http://127.0.0.1:${port}/?lessons=off`);
  await p.waitForTimeout(2500);
  await p.evaluate((v) => window.__play('2-6', v), layout);
  await p.waitForTimeout(4000);
  await p.waitForFunction((E) => { const e = eval(E); return !!(e && e.currentStage && e.player && e.platforms); }, E, { timeout: 60000 });
  await p.waitForFunction(() => document.documentElement.dataset.hudCard !== '1', null, { timeout: 30000 }).catch(() => {});
  await p.waitForTimeout(3000);
  const info = await p.evaluate(([E, K, side]) => {
    const e = eval(E);
    e.isPaused = true;
    const g = e.currentStage.gen2;
    const solid = e.platforms.filter((q) => q.type === 'solid' && !q.broken);
    const pits = [];
    for (const d of g.drops ?? []) {
      if (d.ends.kind !== 'floor' || d.ends.y === undefined) continue;
      const bough = solid.find((q) => Math.abs(q.y - d.ends.y) <= 1 && q.style === 'branch' && q.x <= d.x0 + 2 && q.x + q.width >= d.x1 - 2);
      if (!bough) continue;
      const faces = e.platforms.filter((q) => q.style === 'moss_stone' && q.y + q.height >= bough.y && q.y + q.height <= bough.y + bough.height + 2 && (Math.abs(q.x + q.width - bough.x) < 3 || Math.abs(q.x - (bough.x + bough.width)) < 3));
      const lipL = solid.filter((q) => q !== bough && Math.abs(q.x + q.width - bough.x) < 3 && q.y < bough.y && q.y >= bough.y - 100).sort((a, c) => a.y - c.y)[0];
      const lipR = solid.filter((q) => q !== bough && Math.abs(q.x - (bough.x + bough.width)) < 3 && q.y < bough.y && q.y >= bough.y - 100).sort((a, c) => a.y - c.y)[0];
      pits.push({ d, bough: { x: bough.x, y: bough.y, w: bough.width, h: bough.height }, faces: faces.map((q) => ({ x: q.x, y: q.y, w: q.width, h: q.height, style: q.style })), lipL: lipL && { x: lipL.x, y: lipL.y, w: lipL.width, h: lipL.height, style: lipL.style }, lipR: lipR && { x: lipR.x, y: lipR.y, w: lipR.width, h: lipR.height, style: lipR.style } });
    }
    pits.sort((a, c) => c.bough.y - a.bough.y);
    const pit = pits[K];
    if (!pit) return { pits: pits.length };
    // (near: on the left lip's street, 40 px before the pit, facing right; far: on the right lip's street, facing left)
    const lip = side === 'near' ? (pit.lipL ?? pit.lipR) : (pit.lipR ?? pit.lipL);
    const fromLeft = lip === pit.lipL;
    const pw = e.player.width, ph = e.player.height;
    const x = fromLeft ? pit.bough.x - 40 - pw : pit.bough.x + pit.bough.w + 40;
    Object.assign(e.player, { x, y: lip.y - ph - 2, vx: 0, vy: 0, grounded: false, hanging: false, climbing: false, facing: fromLeft ? 1 : -1, invincibleTimer: 9 });
    for (const hh of e.currentStage?.proto?.hints ?? []) hh.shown = true;
    for (let i = 0; i < 60; i++) e.update(1 / 30);
    return { pits: pits.length, pit, seed: g.seed, rerolls: g.rerolls, built: g.budget?.difficulty, player: { x: e.player.x, y: e.player.y, w: pw, h: ph, grounded: e.player.grounded } };
  }, [E, K, side]);
  if (!info.pit) { console.log(diff, tag, 'NO PIT', JSON.stringify(info)); await ctx.close(); continue; }
  for (let i = 0; i < 30; i++) {
    await p.evaluate((E) => { const e = eval(E); e.update(1 / 30); e.clearGuide?.(); e.player.invincibleTimer = 0; e.render(); }, E);
    await p.evaluate(() => new Promise((r) => requestAnimationFrame(() => r(0))));
  }
  await p.evaluate((E) => { const e = eval(E); e.player.invincibleTimer = 0; e.render(); }, E);
  const cam = await p.evaluate((E) => { const e = eval(E); return { x: e.camera.x, y: e.camera.y, z: e.zoom, px: e.player.x, py: e.player.y, g: e.player.grounded }; }, E);
  const base = `face-2-6-L${layout}-${diff}-pit${K}-${side}-${tag}`;
  const wide = path.join(out, `${base}-wide.jpg`);
  await p.screenshot({ path: wide, type: 'jpeg', quality: 80 });
  // (the close cut: the pit, both lips 70 px out, Pokey's height over the street and the bough's underside under it)
  const B = info.pit.bough;
  const top = Math.min(info.pit.lipL?.y ?? B.y, info.pit.lipR?.y ?? B.y) - 150, bot = B.y + B.h + 50;
  const x0 = B.x - 110, x1 = B.x + B.w + 110;
  let cx = (x0 - cam.x) * cam.z, cy = (top - cam.y) * cam.z, cw = (x1 - x0) * cam.z, ch = (bot - top) * cam.z;
  const L = Math.max(0, cx), T = Math.max(0, cy);
  cw = Math.min(w, cx + cw) - L; ch = Math.min(h, cy + ch) - T;
  const close = path.join(out, `${base}-close.jpg`);
  if (cw > 20 && ch > 20) await p.screenshot({ path: close, type: 'jpeg', quality: 85, clip: { x: L, y: T, width: cw, height: ch } });
  console.log(JSON.stringify({ diff, tag, seed: info.seed, rerolls: info.rerolls, built: info.built, stored: diff, softPits: info.pits, pit: info.pit, placed: info.player, settled: cam, wide, close: cw > 20 && ch > 20 ? close : 'off screen' }));
  await ctx.close();
}
await b.close();
