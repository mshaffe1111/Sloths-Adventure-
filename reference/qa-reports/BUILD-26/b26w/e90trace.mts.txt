const orig = RealPlay.prototype.frame;
RealPlay.prototype.frame = function (this: any, pad: any) {
  const out = orig.call(this, pad);
  T = ++n / 30;
  if (T >= t0 && T <= t1) {
    const p = this.p; const so = this.standingOn; const sq = so != null ? (this.platforms ?? []).find((q: any) => q.id === so) : undefined;
    const keys = pad ? Object.entries(pad).filter(([, v]) => v).map(([k, v]) => (v === true ? k : `${k}=${v}`)).join(' ') : '';
    console.log(`F ${T.toFixed(2)} on ${so ?? '-'}${sq?.wheel ? ' bucket#' + sq.wheel.i : ''} x ${p.x.toFixed(0)} y ${p.y.toFixed(0)} vx ${(p.vx ?? 0).toFixed(0)} vy ${(p.vy ?? 0).toFixed(0)} g ${p.grounded ? 1 : 0} | ${keys}`);
    if (!dumped) { dumped = true;
      for (const q of this.platforms) if (q.y > 4200 && q.y < 5200 && q.x < 1600 && !q.wheel) console.log(`P ${q.id} ${q.type} ${q.style ?? ''} x ${Math.round(q.x)}-${Math.round(q.x + q.width)} y ${Math.round(q.y)}`);
      const wq = this.platforms.find((q: any) => q.wheel && Math.abs(q.y - p.y) < 400);
      if (wq) { const w = wq.wheel.w; console.log('W', JSON.stringify({ cx: w.cx, cy: w.cy, x: w.x, y: w.y, R: w.R, V: w.V, N: w.N, W: w.W, way: w.way, lay: w.lay })); }
    }
  }
  return out;
};
