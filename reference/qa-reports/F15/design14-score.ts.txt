// scratch: the revised 1-4's beats (the vine field) through the meter's own arithmetic; novelty carried from 1-1..1-3 as built
import { combine, rawOf, scoreStage, fittsBits, landBits, windowBits, OTHER_BITS, STAKES, type Challenge } from './score.ts';
const SCALE = 0.30654;
type Row = [beat: string, t: number, kind: string, bits: number, stakes: number, threat?: boolean];
const jump = (share: number, land = 200, crumble = false, dbl = false) => fittsBits(share) + landBits(land) + (crumble ? 1 : 0) + (dbl ? OTHER_BITS.second : 0);
const pod = (cycle: number, open: number) => windowBits(cycle, open);
function run(name: string, setting: 'normal' | 'easy', rows: Row[], minutes: number) {
  const seen = new Map<string, number>();
  for (const id of ['1-1', '1-2', '1-3']) scoreStage(id, setting, 1, seen);
  const out: Challenge[] = rows.map(([note, t, kind, bits, stakes, threat]) => ({ t, x: t * 200, kind, bits, cost: 'retry' as any, stakes, threat: !!threat, overlap: 1, novelty: 1, points: 0, note }));
  const r = combine(out, minutes, seen);
  const by = new Map<string, number>();
  for (const c of out) by.set(c.note, (by.get(c.note) ?? 0) + c.points);
  console.log(`${name}: meter ${(rawOf(r.mean, r.peak) * SCALE).toFixed(1)}  points ${r.points.toFixed(1)}  mean ${r.mean.toFixed(1)}/min  peak ${r.peak.toFixed(1)}/min`);
  console.log('  ' + [...by].map(([k, v]) => `${k} ${v.toFixed(1)}`).join(' | '));
}
