// Each option's core idea, run on real code (designer, 2026-11-04).
import { rhythmLayout } from './src/game/w1Rhythm.ts';
import { traceNode, checkTrace, easyNode, type NodeSpec, type Step, type Move } from './src/game/beatTrace.ts';
import { stageBpm } from './src/game/beatTrace.ts';

const out: string[] = [];
const p = (s: string) => out.push(s);

// ---------- A: are 1-2's real nodes portable across World 1's tempo range? (rooms proven over a range) ----------
p('## A. 1-2 built at every whole BPM of World 1 (126-134), 16 layouts, Normal and Easy');
p('| BPM | Normal: built / 16 (rerolls) | Easy: built / 16 (rerolls) | expert clock Normal (s) |');
p('|---|---|---|---|');
for (let bpm = 126; bpm <= 134; bpm++) {
  const row: string[] = [];
  let clocks: number[] = [];
  for (const easy of [false, true]) {
    let ok = 0, rr = 0;
    for (let seed = 1; seed <= 16; seed++) {
      try { const L = rhythmLayout('1-2', seed, easy, bpm); ok++; rr += L.reroll; if (!easy) clocks.push(L.expert); } catch { /* no layout */ }
    }
    row.push(`${ok}/16 (${rr})`);
  }
  p(`| ${bpm} | ${row[0]} | ${row[1]} | ${clocks.length ? Math.round(Math.min(...clocks)) + '-' + Math.round(Math.max(...clocks)) : '-'} |`);
}

// ---------- B: random step search (WR fitSteps' way) vs a backtracking search that traces prefixes (Tanagra's way) ----------
// A node of 1-2's kind: canopy branches, the palette of WR PAL_A (jumpDy -60..40, hopDy -40..-20).
const JUMP_DY = [-60, -40, -20, 0, 20, 40], HOP_DY = [-40, -30, -20];
const BARS = [['x...x...', 'x..o....', 'x...x...'], ['x..o....', 'x..o....', 'x...x...'], ['h...h...', 'x...x...', 'x..o....'], ['x...x..o', '....x...', 'x...x...']];
let seedR = 12345; const rnd = () => ((seedR = (seedR * 1103515245 + 12345) >>> 0) / 4294967296);
const pick = <T>(a: T[]) => a[Math.floor(rnd() * a.length)];
const marksOf = (bars: string[]) => bars.join('').split('').map((c, i) => (c === '.' ? -1 : i)).filter((i) => i >= 0);
const steps1 = (): Step => (rnd() < 0.35 ? { move: 'hop' as Move, dy: pick(HOP_DY), surface: 'branch' } : { move: 'jump' as Move, dy: pick(JUMP_DY), surface: 'branch' });
function ok(spec: NodeSpec, steps: Step[], easy: boolean): boolean {
  try { const s = { ...spec, steps }; const tr = traceNode(easy ? easyNode(s) : s); return checkTrace(tr, steps, easy).length === 0; } catch { return false; }
}
function prefixSpec(bars: string[], k: number): string[] {
  // keep the first k marks, the rest of the bars run ('.')
  const flat = bars.join('').split(''); let seen = 0;
  for (let i = 0; i < flat.length; i++) if (flat[i] !== '.') { seen++; if (seen > k) flat[i] = '.'; }
  const outB: string[] = []; for (let b = 0; b < bars.length; b++) outB.push(flat.slice(b * 8, b * 8 + 8).join('')); return outB;
}
p('\n## B. Finding steps for a node: random lists (fitSteps) vs backtracking on traced prefixes (Tanagra-style)');
p('| node bars | setting | random: traces until a fit (mean of 20, cap 400) | random: share found within 160 | backtracking: traces until a fit (mean of 20) |');
p('|---|---|---|---|---|');
for (const bars of BARS) {
  for (const easy of [false, true]) {
    const bpm = 128.2; const spec: NodeSpec = { bpm, x: 0, y: 600, bars, steps: [], noOverhang: true };
    const n = marksOf(bars).length;
    let rTot = 0, rWithin = 0, bTot = 0;
    for (let rep = 0; rep < 20; rep++) {
      let t = 0, found = false;
      while (t < 400 && !found) { t++; const st = Array.from({ length: n }, steps1); found = ok(spec, st, easy); }
      rTot += t; if (found && t <= 160) rWithin++;
      // backtracking: extend one step at a time, trying candidates in random order; a prefix must trace before extending
      let traces = 0;
      const cand = (): Step[] => { const c: Step[] = [...JUMP_DY.map((dy) => ({ move: 'jump' as Move, dy, surface: 'branch' as const })), ...HOP_DY.map((dy) => ({ move: 'hop' as Move, dy, surface: 'branch' as const }))]; return c.sort(() => rnd() - 0.5); };
      const dfs = (prefix: Step[]): Step[] | null => {
        if (prefix.length === n) return prefix;
        for (const c of cand()) {
          const next = [...prefix, c]; traces++;
          if (traces > 4000) return null;
          const sp = { ...spec, bars: prefixSpec(bars, next.length) };
          if (!ok(sp, next, easy)) continue;
          const r = dfs(next); if (r) return r;
        }
        return null;
      };
      dfs([]); bTot += traces;
    }
    p(`| ${bars.join(' ')} | ${easy ? 'Easy' : 'Normal'} | ${(rTot / 20).toFixed(1)} | ${rWithin}/20 | ${(bTot / 20).toFixed(1)} |`);
  }
}

// ---------- C: what kinds of node does 1-2 produce over 16 layouts? (an archive's cells, from the layout's own nodes) ----------
p('\n## C. 1-2\'s nodes as archive cells (direction x density band x tag), 16 layouts, Normal');
const cells = new Map<string, number>();
const bpm12 = stageBpm({ worldIndex: 1, id: '1-2', miniIndex: 2 });
for (let seed = 1; seed <= 16; seed++) {
  const L = rhythmLayout('1-2', seed, false, bpm12);
  for (const nd of L.nodes) {
    const band = nd.density === 0 ? 'none' : nd.density < 0.5 ? 'low' : nd.density < 0.9 ? 'mid' : 'high';
    const k = `${nd.dir > 0 ? 'right' : 'left'} | ${band} | ${nd.tag}`; cells.set(k, (cells.get(k) ?? 0) + 1);
  }
}
p(`distinct cells filled: ${cells.size} (of ${2 * 4 * 7} possible); nodes: ${[...cells.values()].reduce((a, b) => a + b, 0)}`);
p('| cell (direction, density band, tag) | nodes |'); p('|---|---|');
[...cells.entries()].sort((a, b) => b[1] - a[1]).forEach(([k, v]) => p(`| ${k} | ${v} |`));
console.log(out.join('\n'));
