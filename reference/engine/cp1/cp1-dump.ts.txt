// CP1 dump: build a stage exactly as engine.ts does (Normal: withGoals(generateStage(cfg, 'normal')); Easy: easyStage(withGoals(generateStage(cfg,
// 'easy')))) with the layout roll set, and print its platforms, vines and ropes as JSON. Run from inside a worktree: npx tsx ../cp1-dump.ts <tag> <ids> <layouts>
import './src/game/headlessDom.ts';
import { generateAllLevels } from './src/game/worldsData.ts';
import { generateStage, setLayoutVariant } from './src/game/stageGenerator.ts';
import { easyStage } from './src/game/easyMode.ts';
import { withGoals } from './src/game/goals.ts';
const [tag, idsArg, nArg] = process.argv.slice(2);
const levels = generateAllLevels();
const out: unknown[] = [];
for (const id of idsArg.split(',')) {
  const c = levels.find((l) => l.id === id)!;
  for (let v = 1; v <= +nArg; v++) for (const setting of ['normal', 'easy'] as const) {
    setLayoutVariant(v);
    const cfg = { ...c, unlocked: true };
    const st = setting === 'easy' ? easyStage(withGoals(generateStage(cfg, 'easy'))) : withGoals(generateStage(cfg, 'normal'));
    setLayoutVariant(null);
    const s = st as unknown as { platforms: any[]; vines?: any[]; ropes?: any[]; enemies?: any[]; waterY?: number; exitTotemX?: number; proto?: any; spawn?: any };
    out.push({
      tag, id, layout: v, setting, waterY: s.waterY ?? null, exitX: s.exitTotemX ?? null,
      proto: !!s.proto, rhythm: !!s.proto?.rhythm,
      platforms: s.platforms.map((p) => ({ x: Math.round(p.x), y: Math.round(p.y), w: Math.round(p.width), h: Math.round(p.height), type: p.type, style: p.style ?? null, deco: !!(p.decorative || p.isDecor || p.ghost), lily: !!(p.lily || p.lilyPad), soil: !!p.soil, creak: p.creak ?? null, pit: p.pit ?? null, catchL: !!p.catch, bare: !!p.bare, drift: !!p.drift, crumble: p.crumbleAfter ?? null, keys: Object.keys(p).filter((k) => !['id', 'x', 'y', 'width', 'height', 'type', 'style'].includes(k)).join('|') })),
      vines: (s.vines ?? []).map((q) => ({ x: Math.round(q.x), top: Math.round(q.topY), bot: Math.round(q.bottomY), type: q.vineType })),
      ropes: (s.ropes ?? []).map((r) => ({ x0: Math.round(r.x0), x1: Math.round(r.x1), y: Math.round(r.y) })),
      enemies: (s.enemies ?? []).map((e) => ({ t: e.type, x: Math.round(e.x), y: Math.round(e.y) })),
      collect: ((st as any).collectibles ?? []).map((c: any) => ({ t: c.type, x: Math.round(c.x), y: Math.round(c.y) })),
      checkpoints: ((st as any).checkpoints ?? []).map((c: any) => ({ x: Math.round(c.x), y: Math.round(c.y) })),
      spawnX: (st as any).spawnX ?? null,
    });
  }
}
process.stdout.write(JSON.stringify(out));
