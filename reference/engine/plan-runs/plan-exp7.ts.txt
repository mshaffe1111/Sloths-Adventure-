// E10: Option E's hang-under, measured from what already ships: the Shimmy Rope (src/game/rope.ts; skills.ts: taught by 1-3, needed from 1-4).
// Grab: his hands reach HAND_REACH px over his feet (rope.ts:14); a rope hangs ROPE_RISE px over the ledges (rope.ts:16). The grab window is
// the engine catches him when his hands are between 12 px over and 30 px under the rope (engine.ts:4983: handY > ry - 12 && handY < ry + 30),
// so his feet must reach rise - HAND_REACH - 30. Reported: the apex, and the time his feet are in that catch band (one 30 fps frame = 33 ms is enough).
import { flight, SHORT_HOLD } from './src/game/beatTrace.ts';
import { PHYS } from './src/game/physics.ts';
import { HAND_REACH, ROPE_RISE, ROPE_SPEED, ROPE_GRAB } from './src/game/rope.ts';
const dt = 1 / 60;
for (const rise of [ROPE_RISE, 120, 160, 180]) {
  const need = rise - HAND_REACH - 30, top = rise - HAND_REACH + 12;
  const row: string[] = [];
  for (const [name, hold] of [['full', 1], ['short flick N', SHORT_HOLD.normal], ['Easy hold', 0.7], ['Easy short', SHORT_HOLD.easy]] as const) {
    const f = flight({ move: 'jump', dy: -200, vx: PHYS.maxRun, hold, dt });
    const pts = f!.path.filter(([, y]) => y >= need && y <= top);
    const ms = pts.length ? Math.round(pts.length * dt * 1000) : 0;
    const apex = Math.max(...f!.path.map(([, y]) => y));
    row.push(`${name} apex ${apex.toFixed(0)} px, grab ${ms} ms`);
  }
  console.log(`rope ${rise} px over the ledge (feet in ${need}-${top}): ${row.join(' | ')}`);
}
console.log(`release onto a drop zone: window = zone / ROPE_SPEED (${ROPE_SPEED} px/s): 30 px ${Math.round(30 / ROPE_SPEED * 1000)} ms, 60 px ${Math.round(60 / ROPE_SPEED * 1000)} ms, 120 px ${Math.round(120 / ROPE_SPEED * 1000)} ms; grab past a rope's end up to ROPE_GRAB ${ROPE_GRAB} px`);
